For developers

Hung in three lines.

Import the exhibits you want and drop them into any page. No keys for the simple ones, one command for the rest.

  • Typed props with autocomplete
  • You only ship the exhibits you import
  • Restyle everything with CSS variables

import { NowPlaying, Streak, LocalTime } from "bragkit"

export default function About() {
  return (
    <section>
      <NowPlaying user="you" />
      <Streak source="github" />
      <LocalTime zone="Europe/Lisbon" />
    </section>
  )
}
